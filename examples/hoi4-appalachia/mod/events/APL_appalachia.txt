# Weekly pulse on APL while the Veil stands: decay, repel invaders, warn, collapse.
country_event = {
	id = appalachia.10
	hidden = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_global_flag = APL_veil_intact }
			if = {
				limit = { NOT = { has_country_flag = APL_eternal_veil } }
				subtract_from_variable = { APL_veil_strength = APL_veil_decay }
			}
			APL_clamp_veil = yes
			APL_repel_invaders = yes
			if = {
				limit = {
					check_variable = { APL_veil_strength < 25 }
					check_variable = { APL_veil_strength > 0 }
					NOT = { has_country_flag = APL_veil_warned }
				}
				set_country_flag = APL_veil_warned
				country_event = { id = appalachia.20 }
			}
			if = {
				limit = { check_variable = { APL_veil_strength > 40 } }
				clr_country_flag = APL_veil_warned
			}
			if = {
				limit = { check_variable = { APL_veil_strength < 0.01 } }
				country_event = { id = appalachia.21 }
			}
			else = {
				country_event = { id = appalachia.10 days = 7 }
			}
		}
	}
}
